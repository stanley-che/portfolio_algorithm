@startuml
title main_refac.cpp - Daily Workflow (TWSE ETL -> Features -> Predict -> SOCP -> Report)

start
:print banner;
:Eigen::setNbThreads(1);
if (_OPENMP?) then (yes)
  :omp_set_num_threads(8);
endif
:mkdir ./OUT & ./advance_parameter_csv;

partition "0) ETL" {
  :choose daily file\n(daily_60d.csv / DAILY_60D.CSV ...);
  :curl_global_init();

  if (daily exists & non-empty?) then (yes)
    :skip TWSE quotes fetch;
  else (no)
    :TwseQuoteDumper.dump_recent_days(opt=60d, throttle=120ms, months_back=4);
    if (fetch failed?) then (yes)
      :curl_global_cleanup();
      stop
    endif
    :daily = "daily_60d.csv";
  endif

  if (meta.csv exists & non-empty?) then (yes)
    :skip meta build;
  else (no)
    :TwseMetaBuilder.build_meta_csv_from_daily(daily -> meta.csv);
    if (failed?) then (yes)
      :curl_global_cleanup();
      stop
    endif
  endif

  if (brokers.csv exists & non-empty?) then (yes)
    :skip brokers fetch;
  else (no)
    :BrokerFetcher(daily).run() -> brokers.csv;
    if (failed?) then (yes)
      :curl_global_cleanup();
      stop
    endif
  endif

  :curl_global_cleanup();
}

partition "Prescan (optional)" {
  :scan daily CSV\n(unique dates, unique symbols);
  :MAX_SYMBOLS = 1070;
  if (N_est > MAX_SYMBOLS?) then (yes)
    :WARN Universe too large\n(cap later in DataLoader);
  endif
}

partition "1) Load & Features" {
  :dl = DataLoader(base_path);
  :dl.config().winsorize_p = 0.02;
  :dl.config().max_symbols = 600;
  :dl.loadDailyPanelAndBuildFeatures();

  if (DataLoader failed?) then (yes)
    stop
  endif

  :sanity checks\n(matrix non-empty & shapes match);
  if (bad shapes / empty?) then (yes)
    stop
  endif

  :T = C.rows, N = C.cols;
  if (symbols.size != N OR T < 3?) then (yes)
    stop
  endif
  :t = max(0, T-2);
}

partition "2) Configs" {
  :RiskConfig rcfg\n(corr_half_life=30, sigma=GK, std_window=30, c_liq=0.6);
  :ProcessingConfig pxcfg\n(ZScoreClip, fill_missing=0);
  :ForecastConfig fcfg\n(linear, lookback=60, half_life=20);
  :BlackSwanConfig bs_cfg\n(m_default=1.0, ... thresholds ...);
  :CostConfig cost_cfg (default);
  :TrainConfig tr\n(ridge, half_life=60);
  :CalibrationConfig cal_cfg;
  :EnhancementConfig enh_cfg;
}

partition "3) Risk / BlackSwan / Stage1 Forecast" {
  :eng = ProcessEngineForecast(dl);
  :risk = eng.build_liquidity_scaled_risk(t, rcfg);
  :bs = eng.black_swan_scale(t, bs_cfg, external_event=false);
  :fo = eng.stage1_forecast(t, pxcfg, fcfg);
  :print rhat_raw summary;
}

partition "4) Predictor" {
  :pred = Predictor(dl, tr, pxcfg, cal_cfg, enh_cfg);
  :pred_out = pred.predict_day(t, TargetType::OC);
  :print rhat_enh summary;
}

partition "5) SOCP Inputs" {
  :setup sweep\n(k_keep=200);
  :pb.rhat_enh = pred_out.rhat_enh;
  :pb.L = risk.L;
  :pb.w0 = 0;
  :pb.W_turn = 1;
  :pb.P0 = 10000;
  :pb.ADV = feat_Liquidity[t]';
  :pb.sigma_GK = feat_GKVol[t]';
  :pb.Imb = feat_Imbalance[t]';
  :pb.BF = feat_BrokerStrength[t]';
  :pb.BIAS = feat_BIAS[t]';
  :pb.TS = feat_TurnoverShare[t]';
  :pb.group = -1;
  :pb.m = bs.m;
  :pb.turn_norm = L1;

  :scfg.debug_on = true;\nscfg.scs.verbose = true;
}

partition "6A) SOCP sweep + Report (first run)" {
  :gen = SocpCandidateGenerator(scfg);
  :cands = gen.generate(pb, sweep);
  if (cands empty?) then (yes)
    stop
  endif
  :best_idx = pick_best_candidate_net(cands);
  :chosen = cands[best_idx];
  :A_socp = P0 * abs(w);
  :side_socp = sign(w);

  :ProcessExporter.dump_to_excel(... advance_parameter.xlsx ...);
  :dump_trade_plan_excel(trade_plan.xlsx);
  :print wrote xlsx paths;
}

note right
你的程式在這裡已經做完一次：
- gen.generate()
- pick best
- dump excel
但下面又再做一次（6B）
end note

partition "6B) Debug + SOCP sweep + Report (second run)" {
  :dbg = DebugReporter(scfg.debug_on);
  :dbg.check_inputs(pb.rhat_enh, pb.L, pb.w0);

  :gen = SocpCandidateGenerator(scfg);
  :cands = gen.generate(pb, sweep);
  if (cands empty?) then (yes)
    stop
  endif
  :best_idx = pick_best_candidate_net(cands);
  :chosen = cands[best_idx];
  :A_socp = P0 * abs(w);
  :side_socp = sign(w);

  :dbg.quick_feas_sanity(\n wmax=0.05,\n group_cap={},\n tau_max=chosen.tau_max,\n W_turn,\n budget=1.0);

  :has_bad(A_socp) or !isfinite(best_net)?; 
  if (bad?) then (yes)
    :WARN NaN/Inf -> Skip XLSX, CSV only;
  endif

  :ProcessExporter.dump_to_excel(... advance_parameter.xlsx ...);
  :dump_trade_plan_csv(trade_plan.csv);

  if (numbers clean?) then (yes)
    :dump_trade_plan_excel(trade_plan.xlsx);
  else (no)
    :skip trade_plan.xlsx;
  endif
}

:print end banner;
stop
